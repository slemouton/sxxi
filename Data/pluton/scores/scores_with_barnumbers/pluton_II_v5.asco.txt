; pluton lib utility functions
 @macro_def @pluton_note($notename,$vel,$dur)
    {
        if ($notename != "0")
         {
			;	print $notename,@symb2midicent($notename)
			tnote $notename $vel $dur			;conversions in MAX
			@makenote (tnote2,@/(@symb2midicent($notename),100),$vel,$dur)
     	}  
     }

 @macro_def @makenote($name, $p, $vel, $dur) 
          { 
                group myMakenote
               {
                        $name $p $vel 
                   $dur $name $p 0
               }
          }   

@proc_def pluton_chord($notelist,$vel,$dur)
    {
        forall $notename in $notelist
         {
			tnote $notename $vel $dur			;conversions in MAX
			@makenote (tnote2,@/(@symb2midicent($notename),100),$vel,$dur)
     	}  
     }
