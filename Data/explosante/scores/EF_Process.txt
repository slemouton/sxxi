 @proc_def ::Makenote($pitch, $velocity, $channel, $dur) ; pour jouer le midi, via le receive EF_samples_in
 @abort { $d ms EF_samples_in $p 0 $c } 
{
	@local $p, $v, $c, $d ; il faut faire ça parce que sinon on a des pb dans les séquences rapides 
	$p := $pitch 
	$v := $velocity
	$c := $channel 
	$d := $dur
	group
	{
		EF_samples_in $p $v $c
		$d ms EF_samples_in $p 0 $c
	}
}
@proc_def ::MakeChord($pitchlist, $velocity, $channel, $dur) ; cas des accords, c'est plus simple de séparer les cas
{
@local $p
	forall $p in $pitchlist
		{
		::Makenote($p, $velocity, $channel, $dur)
		}
}

@proc_def ::Spat_Rotation($source, $step, $period)
{
	$Spat_Rotation := $MYSELF

@fun_def pitch_chords($obj) ; Pour tous sauf slant et ranrep
	{	
		return $obj.$pitch_in
	}
@fun_def random_pitch_from_tab($obj) ; pour ranrep
	{
		@local $PT
		$PT := [($obj.$pitch_in[@rand_int(@size($obj.$pitch_in))])]
		return $PT
	}
@fun_def transpose_pitch($obj) ; pour Slant
	{
		@local $PT ; il faut passer par un tableau pour pouvoir gérer le cas accord pour wobble
	if (@abs($obj.$current_pitch[0] - $obj.$pitch_in[0])< ($obj.$pitch_in[1]-1)) ;
		{
		$PT := [($obj.$current_pitch[0] + $obj.$pitch_in[2])] ; -1 (descendant) ou 1 (ascendant)
		return $PT
		}		
		else { return 0 }
	}
;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
@fun_def increment_period_cycle($count, $obj)  ; Pour tous sauf ranrep et wobble
	{
	@local $i
	$i := $obj.$count%(@size($obj.$cycle))
	return ($obj.$base_period * ($obj.$cycle[$i]))
	}
@fun_def random_period_from_tab($obj) ; pour ranrep
	{
	return ($obj.$base_period[@rand_int(@size($obj.$base_period))]* ($obj.$cycle)) 
	}

@fun_def compute_period_wobble($obj) ; pour wobble
	{
	return (($obj.$base_period[0]) + @rand_int($obj.$base_period[1] * $obj.$cycle))	
	}

@obj_def EFLoop($mode, $pitch_in, $cycle, $base_period, $chan, $dur, $accvel, $noaccvel, $accproba) 
{
	@local $current_period, $current_pitch, $rand, $vel, $next_period, $next_pitch, $body, $curve, $x, $go, $AccentPitch, $IsAccent, $count
	@init
	 {
	 	$body := 0
	 	$count := 0
	 	switch ($mode) ; on regarde le mode,  et on choisit la fonction next_pitch en conséquence; possible de choisir la fonction depuis l'extérieur 
	 	{
	 		case "seq" :  ;Séquenceur : Cas de Chug, ChaCha, Repeat, Seq ... on répète un accord avec un rythme donné (séquence rythmique et tempo)
	 			$current_pitch := $pitch_in ; déjà un tableau
	 			$next_pitch := @pitch_chords()
	 			$current_period := @increment_period_cycle(0, $THISOBJ);
	 			$next_period := @increment_period_cycle($count) ; currification	 			
	 		case "ranrep": ; cas des ranreps : séquenceur aléatoire, on tire au sort des notes et des périodes (mode arpeggio quoi)
	 			$current_pitch := @random_pitch_from_tab($THISOBJ)
	 			$next_pitch := @random_pitch_from_tab() 
	 			$current_period := @random_period_from_tab($THISOBJ)
	 			$next_period := @random_period_from_tab()
	 		case "slant" : ; cas des slant ; même gestion rythmique par contre on fait des gammes
	 			$current_pitch := [$pitch_in[0]]
				$next_pitch := @transpose_pitch()
	 			$current_period := @increment_period_cycle(0,$THISOBJ)   
	 			$next_period := @increment_period_cycle($count) 
	 		case "wobble" : ;cas des wobble ; on joue des accords par contre la gestion des rymes est différente (variation autour d'une période de base)
	 			$current_pitch := $pitch_in ; déjà un tableau
	 			$next_pitch := @pitch_chords()
	 			$current_period := @compute_period_wobble($THISOBJ)
	 			$next_period := @compute_period_wobble() 
	 	}
	}
	@broadcast EFL_Sync($ancestor) ; restart de touts les EFLoop
	{
		if (@exe_child_of($THISOBJ, $ancestor)) ; pour pouvoir faire des sous-groupe par stratégie
		{
		 $go := false
				abort $body
				$body := 0
		$go := true 
		}
	}
	@broadcast EFL_Period($ancestor, $ch_period)
	{ ; variation de la période 
		if (@exe_child_of($THISOBJ, $ancestor))
		{
		$base_period := $ch_period 
		}
	}
		
	@broadcast EFL_Accents($ancestor, $ch_accvel, $ch_noaccvel, $ch_accodds) ; modification des vélocités (accents, bckgnd, proba)
	{
		if (@exe_child_of($THISOBJ, $ancestor))
		{
		$accvel := $ch_accvel
		$noaccvel := $ch_noaccvel
		$accproba := $ch_accodds
		}
	}
	@method_def EFL_Rythm($ch_cycle) ; changements de rythme par instance pour chugs
	{
		$cycle := $ch_cycle 
	}
	@method_def EFL_Pitch($ch_pitch) ; changements de pitch par instance 
	{
		$pitch_in := $ch_pitch ; 
	}
	@whenever ($go)  ;Corps de l'objet, la boucle a proprement parler. les changements de période interviennent à la loop suivante
	{
		$body := 0
		$count := 0
		abort $body
		$body := {
				 loop ef_loop ($current_period) ms
					{
						$rand := @rand(1) ; tirage au sort pour la vélocité
						if ($rand <= $accproba) 
							{ $vel := $accvel}
						else 
							{ $vel := $noaccvel}
					if (($vel>0) && ($current_pitch>0)) 
						{
						::MakeChord($current_pitch, $vel, $chan, $dur) 
						$current_period := $next_period($THISOBJ) ; calcul de la prochaine période (selon le cas num ou tab)
						$current_pitch := $next_pitch($THISOBJ); calcul du pitch (selon le cas num ou tab) on repasse tous les arguments pour vérifier s'il ya eu un changement.
						$EFLSync  := $THISOBJ
						$count := ($count+1)
						}
					}	
				}
	}	
	@whenever ($current_pitch == [0]) ; le pitch 0 arrête l'instance
	{
		$go := false
		abort $body
		$body := 0
	}
	@whenever ($vel == $accvel)
	{
		$IsAccent := 1
		$GAccent := [$THISOBJ, $current_pitch]
	}
	@init
	{	$go := true }		 ; il faut le mettre après 								;
}

;;;; Rain appraît dans T7 s 11
;;;; à partir d'un pitch de base on déroule une suite d'intervalles successifs à un période donnée
@proc_def ::Rain($base_pitch, $period, $intervals, $vel, $chan, $dur)
{
	@local $pitch, $i
	$pitch := $base_pitch
	$i := 0
loop rainloop ($period) ms
{
	::Makenote($pitch, $vel, $chan, $dur)	
	if ($i<(@size($intervals))) ; pour ne pas avoir d'erreur à la dernière
	{
	$pitch := $pitch + ($intervals[$i])
	}
	$i := $i + 1


}
while ($i<=(@size($intervals)))
}

@proc_def ::ShadowPlay($pitch, $vel, $chan, $dur, $period, $minvel, $transpo_list)
{
	@local $c, $rvel, $rpitch, $nrepeat
	$c := 0
	$rvel := $vel
	$nrepeat := @size($transpo_list)
	loop ($period) 
	{
		$rpitch := $pitch + $transpo_list[$c]
		::Makenote($rpitch, $rvel, $chan, $dur)
		$c := $c + 1
		$rvel := @floor(((1-($c/$nrepeat))*100+($c/$nrepeat)*$minvel))
	}
	until ($c>= $nrepeat)
}
